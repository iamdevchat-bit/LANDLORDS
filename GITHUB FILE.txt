                            }

class PortfolioRequest(BaseModel):
    stock_ticker: str = "AAPL"
    land_area_sqft: float = 5000.0
    land_location_score: float = 8.5 # 1.0 to 10.0 scale for infrastructure/location
    land_budget: float = 150000.0
    risk_tolerance: str = "moderate"

# ==========================================
# 1. STOCK PREDICTION MODEL (Classification)
# ==========================================
def train_and_predict_stock(ticker: str):
    stock = yf.Ticker(ticker)
    df = stock.history(period="2y")
    if df.empty:
        raise ValueError(f"No historical stock data found for {ticker}")
        
    # Feature Engineering using pandas
    df['50_MA'] = df['Close'].rolling(window=50).mean()
    df['200_MA'] = df['Close'].rolling(window=200).mean()
    df['Daily_Return'] = df['Close'].pct_change()
    
    # Target: 1 if tomorrow's price is higher than today's, 0 if lower
    df['Target'] = (df['Close'].shift(-1) > df['Close']).astype(int)
    df = df.dropna()
    
    features = ['Close', '50_MA', '200_MA', 'Daily_Return']
    X = df[features]
    y = df['Target']
    
    # Train Random Forest Classifier
    model = RandomForestClassifier(n_estimators=100, random_state=42)
    model.fit(X, y)
    
    # Predict tomorrow's price direction
    latest_features = X.iloc[-1:]
    prediction = model.predict(latest_features)[0]
    confidence = np.max(model.predict_proba(latest_features)) * 100
    
    return {
        "trend": "UP" if prediction == 1 else "DOWN",
        "confidence": round(confidence, 1)
    }

# ==========================================
# 2. LAND VALUATION & TIMING MODEL (Regression)
# ==========================================
def train_and_predict_land(sqft: float, location_score: float):
    # Dataset representing regional land transactions
    np.random.seed(42)
    n_samples = 300
    sample_sqft = np.random.uniform(1000, 10000, n_samples)
    sample_loc = np.random.uniform(1, 10, n_samples)
    
    # Pricing function: Base cost + area factor + location multiplier + market variance
    sample_prices = 20000 + (sample_sqft * 45) + (sample_loc * 18000) + np.random.normal(0, 5000, n_samples)
    
    X_land = pd.DataFrame({'sqft': sample_sqft, 'location_score': sample_loc})
    y_land = sample_prices
    
    # Train Random Forest Regressor
    model_land = RandomForestRegressor(n_estimators=100, random_state=42)
    model_land.fit(X_land, y_land)
    
    # Predict estimated current valuation
    predicted_price = model_land.predict([[sqft, location_score]])[0]
    
    # Calculate projected appreciation rate based on location score
    annual_growth_rate = 0.05 + (location_score / 100) # e.g. score of 8.5 -> 13.5%/year growth
    projected_3yr_price = predicted_price * ((1 + annual_growth_rate) ** 3)
    
    return {
        "predicted_current_price": round(predicted_price, 2),
        "projected_3yr_price": round(projected_3yr_price, 2),
        "expected_annual_growth": round(annual_growth_rate * 100, 1)
    }

# ==========================================
# 3. API ROUTE & LLM ORCHESTRATION
# ==========================================
@app.post("/api/portfolio/analyze")

        stock_res = train_and_predict_stock(req.stock_ticker)

           - Estimated Current Value: ${land_res['predicted_current_price']:,.2f}
           - Projected 3-Year Value: ${land_res['projected_3yr_price']:,.2f} ({land_res['expected_annual_growth']}% annual growth rate)
           - User's Available Budget: ${req.land_budget:,.2f}
        
        Write 3 short, actionable portfolio bullet points:
        - Stock recommendation (Buy/Hold/Sell decision based on the ML trend).
        - Land recommendation (Is it a good buy given their budget? Best timing to hold or sell?).
        - Overall Portfolio Commentary (Asset allocation between liquid stocks vs illiquid land).
        """
