<script>


const basePrice =
    Number(
        localStorage.getItem(
            "basePrice"
        )
    );


const growthRate =
    Number(
        localStorage.getItem(
            "growthRate"
        )
    );


let scenarioBoost = 0;


document.getElementById(
    "current"
).innerText =
    "₹" +
    basePrice.toLocaleString(
        "en-IN"
    );


function calculateValue(year) {

    const years =
        year - 2026;

    const adjustedRate =
        growthRate +
        scenarioBoost / 100;

    return (
        basePrice *
        Math.pow(
            1 + adjustedRate,
            years
        )
    );

}


function updateFuture() {


    const year =
        Number(
            document.getElementById(
                "yearSlider"
            ).value
        );


    document.getElementById(
        "year"
    ).innerText =
        year;


    const value =
        calculateValue(year);


    document.getElementById(
        "futureValue"
    ).innerText =
        "₹" +
        Math.round(value)
        .toLocaleString(
            "en-IN"
        );


    const percentage =
        (
            (value - basePrice)
            / basePrice
        ) * 100;


    document.getElementById(
        "percentage"
    ).innerText =
        "+" +
        percentage.toFixed(1) +
        "% projected change";


    updateScenario();

}


function toggleScenario(
    element,
    boost
) {


    element.classList.toggle(
        "active"
    );


    if (
        element.classList.contains(
            "active"
        )
    ) {

        scenarioBoost += boost;

    } else {

        scenarioBoost -= boost;

    }


    updateScenario();

}


function updateScenario() {


    const value =
        calculateValue(2035);


    document.getElementById(
        "scenarioValue"
    ).innerText =
        "₹" +
        Math.round(value)
        .toLocaleString(
            "en-IN"
        );


    document.getElementById(
        "final"
    ).innerText =
        "₹" +
        Math.round(value)
        .toLocaleString(
            "en-IN"
        );

}


updateFuture();
